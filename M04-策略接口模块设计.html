M04: 策略接口模块 (waleo-policy-base) 设计文档

优先级: P0 (策略核心) 依赖: M01 (waleo-utils), M02 (waleo-config) 状态: 设计中

1. 模块概述

模块名称: waleo-policy-base 版本: v0.1.0 描述: 提供机器人学习策略的统一接口抽象，包括策略基类（BasePolicy）、归一化工具、策略工厂和配置系统。所有具体策略实现（ACT、Diffusion 等）都必须继承此模块提供的基类。

2. 模块职责

2.1 核心职责

  • 策略基类（BasePolicy）：定义所有策略必须实现的抽象接口
  • 归一化工具：提供输入/输出的归一化和反归一化
  • 策略工厂：注册和创建策略实例
  • 基础配置类：定义策略配置的数据结构
  • 特征类型系统：定义观察和动作的特征类型
  • 策略评估指标：提供通用的评估指标计算

2.2 非职责

  • 不涉及具体策略模型实现（属于 M05 ACT、M06 Diffusion 等）
  • 不涉及训练循环逻辑（属于 M13 训练模块）
  • 不涉及数据加载（属于 M03 数据集模块）

4.2 策略配置基类 (base/config.py)

4.2.1 PolicyConfig 类

from dataclasses import dataclass, field
from typing import Any

@dataclass
class PolicyConfig:
    """策略配置基类

    所有具体策略的配置类都应该继承此类
    """
    # 基础配置
    name: str = "base_policy"                # 策略名称
    input_shapes: dict[str, tuple] = field(default_factory=dict)   # 输入形状
    output_shapes: dict[str, tuple] = field(default_factory=dict)  # 输出形状

    # 归一化配置
    normalization_mapping: dict[str, NormalizationMode] = field(
        default_factory=lambda: {
            "observation.state": NormalizationMode.MEAN_STD,
            "observation.image": NormalizationMode.IDENTITY,
            "action": NormalizationMode.MIN_MAX,
        }
    )

    # 设备配置
    device: str = "cuda"                      # 设备 (cuda/mps/cpu)

    # 其他配置
    kwargs: dict[str, Any] = field(default_factory=dict)  # 其他参数

    def __post_init__(self):
        """配置验证"""
        if self.device not in ["cuda", "mps", "cpu"]:
            raise ValueError(f"Invalid device: {self.device}")

4.3 策略基类 (base/policy.py)

4.3.1 BasePolicy 抽象类

import torch
import torch.nn as nn
from abc import ABC, abstractmethod
from typing import Any
from pathlib import Path

class BasePolicy(nn.Module, ABC):
    """策略基类

    所有具体策略实现（ACT, Diffusion 等）必须继承此类并实现所有抽象方法

    子类必须定义：
    - config_class: 配置类
    - name: 策略名称
    """

    # 子类必须定义
    config_class: type[PolicyConfig]
    name: str

    def __init__(self, config: PolicyConfig):
        """
        Args:
            config: 策略配置
        """
        super().__init__()
        self.config = config
        self.device = torch.device(config.device)

    @abstractmethod
    def get_optim_params(self) -> list[dict]:
        """返回优化器参数组

        用于创建优化器，支持不同参数组使用不同学习率

        Returns:
            参数组列表，例如：
            [
                {"params": model.encoder.parameters(), "lr": 1e-4},
                {"params": model.decoder.parameters(), "lr": 1e-5}
            ]
        """
        pass

    @abstractmethod
    def reset(self):
        """重置策略状态

        在每个 episode 开始时调用，用于清除缓存、重置隐藏状态等
        """
        pass

    @abstractmethod
    def forward(self, batch: dict) -> tuple[torch.Tensor, dict]:
        """训练前向传播

        计算损失并返回日志信息

        Args:
            batch: 批量数据，包含：
                - observation: 观察数据
                - action: 动作数据
                - 其他训练所需数据

        Returns:
            loss: 损失张量
            logs: 日志字典，例如 {"loss": 0.5, "lr": 0.001}
        """
        pass

    @abstractmethod
    def select_action(self, batch: dict) -> torch.Tensor:
        """选择动作（推理）

        Args:
            batch: 批量数据，包含 observation

        Returns:
            action: 动作张量，形状 (batch_size, action_dim)
        """
        pass

    def save(self, path: Path | str):
        """保存策略

        Args:
            path: 保存路径
        """
        path = Path(path)
        path.mkdir(parents=True, exist_ok=True)

        # 保存模型
        torch.save({
            "model_state_dict": self.state_dict(),
            "config": self.config,
        }, path / "policy.safetensors")

    @classmethod
    def load(cls, path: Path | str):
        """加载策略

        Args:
            path: 加载路径

        Returns:
            策略实例
        """
        path = Path(path)
        checkpoint = torch.load(path / "policy.safetensors")

        # 创建实例
        config = checkpoint["config"]
        policy = cls(config)

        # 加载权重
        policy.load_state_dict(checkpoint["model_state_dict"])
        policy.eval()

        return policy

    def eval_action(self, batch: dict) -> dict:
        """评估动作（包含元数据）

        Args:
            batch: 批量数据

        Returns:
            包含 action 和其他信息的字典
        """
        with torch.no_grad():
            action = self.select_action(batch)

        return {
            "action": action.cpu().numpy(),
            "success": True,
        }

使用示例

import torch
import torch.nn as nn
from waleo_policy.base import BasePolicy, PolicyConfig

class MyPolicy(BasePolicy):
    """自定义策略示例"""

    name = "my_policy"
    config_class = PolicyConfig

    def __init__(self, config: PolicyConfig):
        super().__init__(config)
        # 构建网络
        self.network = nn.Sequential(
            nn.Linear(10, 64),
            nn.ReLU(),
            nn.Linear(64, 6)
        )

    def get_optim_params(self):
        return [{"params": self.parameters(), "lr": 1e-4}]

    def reset(self):
        # 清除缓存等
        pass

    def forward(self, batch):
        obs = batch["observation.state"]
        action = batch["action"]

        pred_action = self.network(obs)
        loss = nn.functional.mse_loss(pred_action, action)

        return loss, {"loss": loss.item()}

    def select_action(self, batch):
        obs = batch["observation.state"]
        return self.network(obs)

# 使用
config = PolicyConfig(
    name="my_policy",
    input_shapes={"observation.state": (10,)},
    output_shapes={"action": (6,)}
)
policy = MyPolicy(config)

# 训练
batch = {
    "observation.state": torch.randn(32, 10),
    "action": torch.randn(32, 6)
}
loss, logs = policy(batch)
loss.backward()

# 推理
policy.reset()
action = policy.select_action({"observation.state": torch.randn(1, 10)})

4.4 归一化工具 (normalize/)

4.4.1 Normalize 类 (normalize/normalize.py)

import torch
import torch.nn as nn
from typing import dict

class Normalize(nn.Module):
    """输入归一化

    对观察数据进行归一化处理
    """

    def __init__(
        self,
        input_shapes: dict[str, tuple],
        normalization_mapping: dict[str, NormalizationMode],
        stats: dict[str, dict] | None = None
    ):
        """
        Args:
            input_shapes: 输入形状映射
            normalization_mapping: 归一化模式映射
            stats: 统计信息，格式：
                {
                    "observation.state": {
                        "mean": np.array([...]),
                        "std": np.array([...])
                    }
                }
        """
        super().__init__()
        self.input_shapes = input_shapes
        self.normalization_mapping = normalization_mapping

        # 注册统计信息为 buffer
        self.register_buffers("means", {})
        self.register_buffers("stds", {})

        if stats is not None:
            for key, stat in stats.items():
                if "mean" in stat:
                    self.means[key] = torch.from_numpy(stat["mean"]).float()
                if "std" in stat:
                    self.stds[key] = torch.from_numpy(stat["std"]).float()

    def forward(self, batch: dict) -> dict:
        """对 batch 进行归一化

        Args:
            batch: 输入 batch

        Returns:
            归一化后的 batch
        """
        normalized_batch = {}

        for key, value in batch.items():
            if key in self.normalization_mapping:
                mode = self.normalization_mapping[key]

                if mode == NormalizationMode.MEAN_STD:
                    # (x - mean) / std
                    mean = self.means.get(key, 0)
                    std = self.stds.get(key, 1)
                    value = (value - mean) / (std + 1e-8)

                elif mode == NormalizationMode.MIN_MAX:
                    # 映射到 [-1, 1]
                    # 需要从 stats 中获取 min/max
                    # 这里简化处理，假设数据已标准化
                    value = torch.tanh(value)

                # IDENTITY: 不做处理

            normalized_batch[key] = value

        return normalized_batch

4.4.2 Unnormalize 类 (normalize/unnormalize.py)

import torch
import torch.nn as nn

class Unnormalize(nn.Module):
    """输出反归一化

    将模型输出反归一化到原始动作空间
    """

    def __init__(
        self,
        output_shapes: dict[str, tuple],
        normalization_mapping: dict[str, NormalizationMode],
        stats: dict[str, dict] | None = None
    ):
        """
        Args:
            output_shapes: 输出形状映射
            normalization_mapping: 归一化模式映射
            stats: 统计信息
        """
        super().__init__()
        self.output_shapes = output_shapes
        self.normalization_mapping = normalization_mapping

        # 注册统计信息为 buffer
        self.register_buffers("means", {})
        self.register_buffers("stds", {})

        if stats is not None:
            for key, stat in stats.items():
                if "mean" in stat:
                    self.means[key] = torch.from_numpy(stat["mean"]).float()
                if "std" in stat:
                    self.stds[key] = torch.from_numpy(stat["std"]).float()

    def forward(self, batch: dict) -> dict:
        """对 batch 进行反归一化

        Args:
            batch: 模型输出 batch

        Returns:
            反归一化后的 batch
        """
        unnormalized_batch = {}

        for key, value in batch.items():
            if key in self.normalization_mapping:
                mode = self.normalization_mapping[key]

                if mode == NormalizationMode.MEAN_STD:
                    # x * std + mean
                    mean = self.means.get(key, 0)
                    std = self.stds.get(key, 1)
                    value = value * (std + 1e-8) + mean

                elif mode == NormalizationMode.MIN_MAX:
                    # 从 [-1, 1] 映射回原始范围
                    # 简化处理
                    value = torch.atanh(torch.clamp(value, -0.99, 0.99))

                # IDENTITY: 不做处理

            unnormalized_batch[key] = value

        return unnormalized_batch

4.5 策略工厂 (factory/factory.py)

4.5.1 PolicyFactory 类

from typing import type, dict

class PolicyFactory:
    """策略工厂

    用于注册和创建策略实例
    """

    # 注册表：策略名称 -> 策略类
    _registry: dict[str, type[BasePolicy]] = {}

    @staticmethod
    def register(name: str, cls: type[BasePolicy]):
        """注册策略类型

        Args:
            name: 策略名称
            cls: 策略类
        """
        if not issubclass(cls, BasePolicy):
            raise TypeError(f"{cls} must be a subclass of BasePolicy")

        PolicyFactory._registry[name] = cls

    @staticmethod
    def create(name: str, config: PolicyConfig, **kwargs) -> BasePolicy:
        """创建策略实例

        Args:
            name: 策略名称
            config: 策略配置
            **kwargs: 额外参数

        Returns:
            策略实例

        Raises:
            ValueError: 如果策略名称未注册
        """
        if name not in PolicyFactory._registry:
            raise ValueError(f"Unknown policy: {name}. Available: {list(PolicyFactory._registry.keys())}")

        policy_cls = PolicyFactory._registry[name]
        return policy_cls(config, **kwargs)

    @staticmethod
    def list_policies() -> list[str]:
        """列出所有已注册的策略

        Returns:
            策略名称列表
        """
        return list(PolicyFactory._registry.keys())

    @staticmethod
    def is_registered(name: str) -> bool:
        """检查策略是否已注册

        Args:
            name: 策略名称

        Returns:
            是否已注册
        """
        return name in PolicyFactory._registry

4.6 评估指标 (utils/metrics.py)

4.6.1 指标计算函数

import numpy as np
from typing import dict

def compute_success_rate(episode_returns: list[float], success_threshold: float = 0.0) -> float:
    """计算成功率

    Args:
        episode_returns: 每个 episode 的回报
        success_threshold: 成功阈值

    Returns:
        成功率 (0-1)
    """
    successes = [r > success_threshold for r in episode_returns]
    return np.mean(successes)

def compute_average_return(episode_returns: list[float]) -> float:
    """计算平均回报

    Args:
        episode_returns: 每个 episode 的回报

    Returns:
        平均回报
    """
    return np.mean(episode_returns)

def compute_std_return(episode_returns: list[float]) -> float:
    """计算回报标准差

    Args:
        episode_returns: 每个 episode 的回报

    Returns:
        回报标准差
    """
    return np.std(episode_returns)

def compute_metrics(episode_returns: list[float], success_threshold: float = 0.0) -> dict:
    """计算所有评估指标

    Args:
        episode_returns: 每个 episode 的回报
        success_threshold: 成功阈值

    Returns:
        指标字典
    """
    return {
        "success_rate": compute_success_rate(episode_returns, success_threshold),
        "average_return": compute_average_return(episode_returns),
        "std_return": compute_std_return(episode_returns),
        "max_return": np.max(episode_returns),
        "min_return": np.min(episode_returns),
        "num_episodes": len(episode_returns),
    }

7. 使用场景

7.1 实现自定义策略

import torch
import torch.nn as nn
from waleo_policy.base import BasePolicy, PolicyConfig, NormalizationMode
from waleo_policy.factory import PolicyFactory

class SimplePolicy(BasePolicy):
    """简单的 MLP 策略"""

    name = "simple"
    config_class = PolicyConfig

    def __init__(self, config: PolicyConfig):
        super().__init__(config)

        # 获取输入/输出维度
        obs_dim = config.input_shapes["observation.state"][0]
        action_dim = config.output_shapes["action"][0]

        # 构建网络
        self.network = nn.Sequential(
            nn.Linear(obs_dim, 256),
            nn.ReLU(),
            nn.Linear(256, 256),
            nn.ReLU(),
            nn.Linear(256, action_dim)
        )

    def get_optim_params(self):
        return [{"params": self.parameters(), "lr": 1e-4}]

    def reset(self):
        pass

    def forward(self, batch):
        obs = batch["observation.state"]
        action = batch["action"]

        pred_action = self.network(obs)
        loss = nn.functional.mse_loss(pred_action, action)

        return loss, {"loss": loss.item()}

    def select_action(self, batch):
        obs = batch["observation.state"]
        return self.network(obs)

# 注册策略
PolicyFactory.register("simple", SimplePolicy)

# 使用
config = PolicyConfig(
    name="simple",
    input_shapes={"observation.state": (10,)},
    output_shapes={"action": (6,)}
)
policy = PolicyFactory.create("simple", config)
