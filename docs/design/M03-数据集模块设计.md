> **优先级**: P0 (核心基础)

---

## 1. 模块概述

**模块名称**: `waleo-dataset`
**版本**: v0.1.0
**描述**: 提供机器人学习数据集的加载、预处理、采样和统计功能，支持视频和图像存储格式。

---

## 2. 模块职责

### 2.1 核心职责
- 数据集加载（从本地或 HuggingFace Hub）
- 视频/图像解码和编码
- Episode 级别的数据采样
- 数据统计计算
- 数据变换和预处理
- 多数据集联合加载

### 2.2 非职责
- 不涉及模型训练逻辑（属于 M13 训练模块）
- 不涉及数据记录（属于独立的数据收集工具）
- 不涉及环境交互（属于 M11 环境模块）

## 4. API 设计

### 4.1 核心数据集类 (`core/dataset.py`)

#### LeRobotDataset 类

```python
class LeRobotDataset(torch.utils.data.Dataset):
    """LeRobot 机器人学习数据集

    支持：
    - 从 HuggingFace Hub 加载
    - 本地文件加载
    - 视频/图像存储
    - 时间戳索引
    - Episode 边界感知
    """

    def __init__(
        self,
        repo_id: str,
        root: str | Path | None = None,
        episodes: list[int] | None = None,
        image_transforms: Any | None = None,
        revision: str | None = None,
        video_backend: str = "pyav",
        delta_timestamps: dict | None = None,
        skip_validation: bool = False,
    ):
        """
        Args:
            repo_id: HuggingFace Hub 仓库 ID (如 "waleo/pusht")
            root: 本地存储路径，默认为 HF_LEROBOT_HOME/repo_id
            episodes: 要使用的 episode 索引列表，None 表示全部
            image_transforms: 图像变换配置
            revision: Hub 版本
            video_backend: 视频解码后端 ("pyav", "torchcodec")
            delta_timestamps: 时间戳偏移配置
            skip_validation: 跳过数据验证
        """
        ...

    def __len__(self) -> int:
        """返回数据集总帧数"""
        ...

    def __getitem__(self, idx: int) -> dict:
        """获取单帧数据

        Returns:
            包含以下键的字典：
            - observation.state: 机器人状态
            - observation.image: 相机图像 (如果存在)
            - action: 动作
            - task: 任务描述 (如果存在)
            - episode_index: episode 索引
            - frame_index: 帧索引
            - timestamp: 时间戳
        """
        ...

    @property
    def num_episodes(self) -> int:
        """Episode 总数"""
        ...

    @property
    def fps(self) -> int:
        """数据采集帧率"""
        ...

    @property
    def shape(self) -> dict:
        """各特征的形状"""
        ...

    def get_episode_data(self, episode_index: int) -> dict:
        """获取整个 episode 的数据"""
        ...

    def get_stats(self) -> dict:
        """获取数据集统计信息"""
        ...

    def save_to_disk(self, path: str | Path) -> None:
        """保存数据集到磁盘"""
        ...
```

#### 使用示例

```python
from lerobot_dataset import LeRobotDataset, load_dataset

# 从 Hub 加载
dataset = LeRobotDataset(
    repo_id="waleo/pusht",
    episodes=[0, 1, 2],  # 只使用前 3 个 episodes
    video_backend="pyav"
)

# 访问数据
print(f"Total frames: {len(dataset)}")
print(f"Episodes: {dataset.num_episodes}")
print(f"FPS: {dataset.fps}")

# 获取单帧
frame = dataset[0]
print(frame.keys())
# dict_keys(['observation.state', 'observation.image', 'action', ...])

# 获取统计信息
stats = dataset.get_stats()
print(stats['observation.state']['mean'])
```

### 4.4 数据采样 (`sampling/`)

#### 4.4.1 EpisodeAwareSampler

```python
class EpisodeAwareSampler(torch.utils.data.Sampler):
    """Episode 边界感知采样器

    特点：
    - 尊重 episode 边界
    - 支持丢弃首尾帧
    - 支持 episode 过滤
    - 支持随机打乱
    """

    def __init__(
        self,
        episode_data_index: dict,
        episode_indices: list[int] | None = None,
        drop_n_first_frames: int = 0,
        drop_n_last_frames: int = 0,
        shuffle: bool = False,
    ):
        """
        Args:
            episode_data_index: episode 索引映射
                {"from": tensor([0, 100, 200]), "to": tensor([100, 200, 300])}
            episode_indices: 要使用的 episode 索引
            drop_n_first_frames: 每个 episode 开头丢弃的帧数
            drop_n_last_frames: 每个 episode 结尾丢弃的帧数
            shuffle: 是否打乱顺序
        """
        ...

    def __iter__(self) -> Iterator[int]:
        """生成索引"""
        ...

    def __len__(self) -> int:
        """总采样数"""
        ...
```

#### 使用示例

```python
from lerobot_dataset import LeRobotDataset
from lerobot_dataset.sampling import EpisodeAwareSampler
from torch.utils.data import DataLoader

dataset = LeRobotDataset("waleo/pusht")

# 创建采样器：只使用前 10 个 episodes，每个 episode 丢弃首尾各 5 帧
sampler = EpisodeAwareSampler(
    episode_data_index=dataset.episode_data_index,
    episode_indices=list(range(10)),
    drop_n_first_frames=5,
    drop_n_last_frames=5,
    shuffle=True
)

dataloader = DataLoader(
    dataset,
    batch_size=32,
    sampler=sampler,
    num_workers=4
)
```

## 5. 数据格式规范

### 5.1 目录结构

```
dataset/
├── meta/
│   ├── info.json           # 数据集信息
│   ├── episodes.jsonl      # Episode 元数据
│   ├── stats.json          # 统计信息
│   └── tasks.jsonl         # 任务信息（可选）
├── videos/                 # 视频文件（如果使用视频存储）
│   ├── chunk_0/
│   │   ├── episode_0.mp4
│   │   └── episode_1.mp4
│   └── chunk_1/
│       └── episode_2.mp4
└── data/                   # Parquet 文件
    ├── chunk_0/
    │   ├── episode_0.parquet
    │   └── episode_1.parquet
    └── chunk_1/
        └── episode_2.parquet
```

### 5.2 info.json 格式

```json
{
    "codebase_version": "v2.1",
    "robot_type": "so100",
    "fps": 30,
    "total_episodes": 100,
    "total_frames": 150000,
    "data_path": "data/chunk_{episode_chunk:04d}/episode_{episode_index:06d}.parquet",
    "video_path": "videos/chunk_{episode_chunk:04d}/episode_{episode_index:06d}.mp4",
    "features": {
        "observation.state": {
            "dtype": "float32",
            "shape": [2],
            "names": ["joint_0", "joint_1"]
        },
        "observation.image": {
            "dtype": "video",
            "shape": [3, 480, 640]
        },
        "action": {
            "dtype": "float32",
            "shape": [2],
            "names": ["joint_0", "joint_1"]
        },
        "episode_index": {
            "dtype": "int64",
            "shape": [1]
        },
        "frame_index": {
            "dtype": "int64",
            "shape": [1]
        },
        "timestamp": {
            "dtype": "float64",
            "shape": [1]
        }
    }
}
```

### 5.3 数据帧格式

```python
{
    # 观察数据
    "observation.state": np.array([0.5, 0.3]),           # 关节位置
    "observation.image": np.array((3, 480, 640)),        # 图像 (C,H,W)

    # 动作数据
    "action": np.array([0.6, 0.4]),                     # 动作

    # 元数据
    "episode_index": 0,                                  # Episode 索引
    "frame_index": 100,                                  # 帧 索引
    "timestamp": 3.33,                                   # 时间戳（秒）

    # 可选
    "task": "Pick up the object",                        # 任务描述
    "reward": 1.0,                                       # 奖励
    "next.reward": 0.8,                                  # 下一帧奖励
    "next.done": False,                                  # 是否结束
}
```

## 9. 性能考虑

### 9.1 性能目标
- `__getitem__`: < 50ms (不含视频解码)
- 视频解码: < 10ms/帧 (使用 GPU 加速)
- 批量加载: > 1000 samples/s

### 9.2 优化策略
- 使用视频编码减少存储空间
- 缓存解码后的帧
- 预取和异步加载
- 使用 Delta Timestamps 减少数据量
