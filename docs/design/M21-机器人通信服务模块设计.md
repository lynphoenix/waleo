## 1. 模块概述

**模块名称**: `waleo-robot-comm`
**版本**: v0.1.0
**描述**: 提供机器人与远程训练/推理服务器之间的通信服务，实现异步推理架构。使用 M01 的 RPC 基础设施，支持观察数据上传和动作指令下载。

---

## 2. 模块职责

### 2.1 核心职责
- 机器人端服务（Robot Service）：接收推理请求、返回观察数据
- 服务器端客户端（Server Client）：发送推理请求、接收动作预测
- 基于事件驱动的异步通信
- 数据流管理（观察 → 服务器 → 动作 → 机器人）
- 连接管理和错误恢复
- 多机器人支持

### 2.2 非职责
- 不涉及 RPC 基础设施（属于 M01）
- 不涉及具体策略模型推理（属于推理服务器）
- 不涉及机器人硬件控制（属于 M09）
- 不涉及训练逻辑（属于 M13）

## 4. API 设计

### 4.1 机器人端服务 (`robot/service.py`)

#### RobotService 类

```python
class RobotService(RPCServer):
    """机器人通信服务

    运行在机器人上，提供：
    - 接收推理请求
    - 返回当前观察
    - 执行接收到的动作
    - 流式数据传输
    """

    def __init__(
        self,
        robot,                      # 机器人实例
        host: str = "0.0.0.0",
        port: int = 5000,
        max_workers: int = 4,
        enable_compression: bool = True,
        enable_auth: bool = False,
        **kwargs
    ):
        """
        Args:
            robot: 机器人实例 (来自 M09)
            host: 绑定地址
            port: 绑定端口
            max_workers: 最大工作线程数
            enable_compression: 启用数据压缩
            enable_auth: 启用认证
        """
        super().__init__(host, port, max_workers, **kwargs)
        self.robot = robot
        self.enable_compression = enable_compression
        self.enable_auth = enable_auth

        # 注册 RPC 方法
        self.register_method("get_observation", self.get_observation)
        self.register_method("send_action", self.send_action)
        self.register_method("get_info", self.get_info)
        self.register_method("stream_observation", self.stream_observation)

    def get_observation(self) -> dict:
        """获取当前观察

        Returns:
            observation: 包含状态、图像等
            timestamp: 时间戳
        """
        obs = self.robot.get_observation()
        return {
            "observation": obs,
            "timestamp": time.time()
        }

    def send_action(self, action: dict) -> dict:
        """发送动作给机器人执行

        Args:
            action: 动作指令
                - action: 动作数据
                - repeat: 重复次数（用于 action chunking）

        Returns:
            success: 是否成功
            message: 状态消息
        """
        try:
            action_data = action["action"]
            repeat = action.get("repeat", 1)

            for _ in range(repeat):
                self.robot.send_action(action_data)
                time.sleep(self.robot.control_frequency)

            return {
                "success": True,
                "message": f"Action executed {repeat} times"
            }
        except Exception as e:
            return {
                "success": False,
                "message": str(e)
            }

    def get_info(self) -> dict:
        """获取机器人信息

        Returns:
            robot_type: 机器人类型
            capabilities: 能力列表
            state: 当前状态
        """
        return {
            "robot_type": self.robot.robot_type,
            "capabilities": self.robot.capabilities,
            "state": self.robot.state
        }

    def stream_observation(self) -> Iterator[dict]:
        """流式传输观察数据

        用于实时监控和在线训练

        Yields:
            observation: 观察数据
            timestamp: 时间戳
        """
        while True:
            obs = self.robot.get_observation()
            yield {
                "observation": obs,
                "timestamp": time.time()
            }
            time.sleep(1.0 / self.robot.camera_fps)

    def start(self, block: bool = True):
        """启动服务

        Args:
            block: 是否阻塞运行
        """
        # 初始化机器人
        self.robot.connect()
        self.robot.reset()

        # 启动 RPC 服务
        super().start(block=block)

    def stop(self):
        """停止服务"""
        super().stop()
        self.robot.disconnect()
```

### 4.2 服务器端客户端 (`server/client.py`)

#### InferenceClient 类

```python
class InferenceClient(RPCClient):
    """推理客户端

    运行在训练/推理服务器上，连接到机器人服务
    """

    def __init__(
        self,
        host: str,
        port: int = 5000,
        timeout: float = 5.0,
        enable_compression: bool = True,
        auto_reconnect: bool = True,
        **kwargs
    ):
        """
        Args:
            host: 机器人服务地址
            port: 服务端口
            timeout: 超时时间
            enable_compression: 启用压缩
            auto_reconnect: 自动重连
        """
        super().__init__(host, port, timeout, **kwargs)
        self.enable_compression = enable_compression
        self.auto_reconnect = auto_reconnect

    def get_observation(self) -> dict:
        """获取机器人观察

        Returns:
            observation: 观察数据
            timestamp: 时间戳
        """
        return self.call("get_observation")

    def send_action(self, action: np.ndarray, repeat: int = 1) -> dict:
        """发送动作给机器人

        Args:
            action: 动作数据
            repeat: 重复次数（用于 action chunking）

        Returns:
            success: 是否成功
            message: 状态消息
        """
        return self.call("send_action", {
            "action": action.tolist() if isinstance(action, np.ndarray) else action,
            "repeat": repeat
        })

    def get_info(self) -> dict:
        """获取机器人信息"""
        return self.call("get_info")

    def stream_observation(self) -> Iterator[dict]:
        """流式接收观察数据"""
        return self.call_stream("stream_observation")

    def predict_and_execute(
        self,
        policy,
        n_steps: int = 100,
        action_chunk_size: int = 10
    ) -> dict:
        """完整的预测-执行循环

        Args:
            policy: 策略模型
            n_steps: 执行步数
            action_chunk_size: 每次预测的动作数

        Returns:
            metrics: 执行指标
        """
        metrics = {
            "n_steps": 0,
            "n_success": 0,
            "n_errors": 0,
            "total_time": 0.0
        }

        start_time = time.time()

        for step in range(0, n_steps, action_chunk_size):
            try:
                # 获取观察
                obs_data = self.get_observation()
                obs = obs_data["observation"]

                # 预测动作 chunk
                actions = policy.select_action(obs, n_actions=action_chunk_size)

                # 发送并执行动作
                for action in actions:
                    result = self.send_action(action, repeat=1)
                    if result["success"]:
                        metrics["n_success"] += 1
                    else:
                        metrics["n_errors"] += 1

                metrics["n_steps"] += len(actions)

            except Exception as e:
                metrics["n_errors"] += 1
                print(f"Error at step {step}: {e}")

                if self.auto_reconnect:
                    self.connect()

        metrics["total_time"] = time.time() - start_time
        return metrics
```

#### 使用示例

```python
from waleo_robot_comm.server import InferenceClient
from waleo_policy_act import ACTPolicy

# 连接到机器人

    host="192.168.1.100",  # 机器人 IP
    port=5000,
    auto_reconnect=True
)
client.connect()

# 获取机器人信息
info = client.get_info()

# 加载策略

# 运行推理

    policy=policy,
    n_steps=1000,
    action_chunk_size=10
)

```

#### 4.5.2 编解码器 (`protocol/codec.py`)

```python
class RobotMessageCodec:
    """机器人消息编解码器

    处理：
    - 观察（包含图像）的序列化/反序列化
    - 动作的序列化/反序列化
    - 压缩和解压缩
    """

    def __init__(self, enable_compression: bool = True):
        self.enable_compression = enable_compression

    def encode_observation(self, obs: dict) -> bytes:
        """编码观察数据

        Args:
            obs: 包含图像和状态的观察

        Returns:
            序列化后的字节数据
        """
        # 使用 waleo_utils 的序列化工具
        from waleo_utils.communication import serialize

        data = serialize(obs)

        if self.enable_compression:
            data = self._compress(data)

        return data

    def decode_observation(self, data: bytes) -> dict:
        """解码观察数据"""
        if self.enable_compression:
            data = self._decompress(data)

        from waleo_utils.communication import deserialize
        return deserialize(data)

    def encode_action(self, action: np.ndarray) -> bytes:
        """编码动作"""
        from waleo_utils.communication import serialize
        return serialize(action)

    def decode_action(self, data: bytes) -> np.ndarray:
        """解码动作"""
        from waleo_utils.communication import deserialize
        return deserialize(data)

    def _compress(self, data: bytes) -> bytes:
        """压缩数据"""
        import zlib
        return zlib.compress(data)

    def _decompress(self, data: bytes) -> bytes:
        """解压缩数据"""
        import zlib
        return zlib.decompress(data)
```

from waleo_robot_comm.server import InferenceClient
from waleo_policy_act import ACTPolicy

client = InferenceClient("192.168.1.100", port=5000)
client.connect()

policy = ACTPolicy.from_pretrained("./checkpoints/so100")

# 运行推理

    n_steps=1000,
    action_chunk_size=10
)

### 7.2 多机器人并行控制

```python
from waleo_robot_comm.server import RobotConnectionPool, LoadBalancer

# 创建连接池
pool = RobotConnectionPool(
    robots=[
        {"id": "robot_0", "host": "192.168.1.100", "port": 5000},
        {"id": "robot_1", "host": "192.168.1.101", "port": 5000},
        {"id": "robot_2", "host": "192.168.1.102", "port": 5000},
    ],
    max_connections_per_robot=2
)

balancer = LoadBalancer(pool, strategy="round_robin")

# 并行执行任务
from concurrent.futures import ThreadPoolExecutor

def execute_on_robot(robot_id, policy):
    with pool.acquire(robot_id) as client:
        return client.predict_and_execute(policy, n_steps=100)

policy = ACTPolicy.from_pretrained("./checkpoints/so100")

with ThreadPoolExecutor(max_workers=3) as executor:
    futures = [
        executor.submit(execute_on_robot, f"robot_{i}", policy)
        for i in range(3)
    ]
    results = [f.result() for f in futures]
```
