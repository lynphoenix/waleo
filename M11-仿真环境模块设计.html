M11: 仿真环境模块 (waleo-env) 设计文档

优先级: P1 (环境与训练) 依赖: M02 (waleo-config) 状态: 设计中

1. 模块概述

模块名称: waleo-env 版本: v0.1.0 描述: 提供 Gym/Gymnasium 兼容的环境基类和仿真环境，用于机器人模仿学习训练和验证。支持多种仿真后端和任务类型。

2. 模块职责

2.1 核心职责

  • Gym/Gymnasium 兼容的环境基类（BaseEnv, VectorEnv, EnvWrapper）：提供标准化的环境接口
  • 机器人环境基类（RobotEnv）：扩展 BaseEnv，添加机器人特有功能
  • 提供具体的仿真环境实现
  • 支持常见的机器人操作任务（推、抓取、放置等）
  • 环境工厂函数，方便创建不同配置的环境
  • 支持多种仿真后端（Isaac Gym, MuJoCo, PyBullet, SAPIEN, ManiSkill）
  • 奖励函数计算
  • 任务终止条件判断
  • 域随机化支持

2.2 非职责

  • 不涉及训练循环逻辑（属于 M13 训练模块）
  • 不涉及策略模型（属于 M04-M06）
  • 不涉及真实机器人控制（属于 M09 机器人模块）

4.1 Gym 兼容的环境基类 (base/)

4.1.1 BaseEnv (base/base.py)

设计理念：提供与 Gym/Gymnasium 兼容的环境基类

from abc import ABC, abstractmethod
from typing import Optional, Tuple, Dict, Any
import numpy as np

class BaseEnv(ABC):
    """环境基类

    与 Gym/Gymnasium API 兼容，提供：
    - 标准的 reset/step 接口
    - 观察和动作空间定义
    - 渲染支持
    """

    @property
    @abstractmethod
    def observation_space(self) -> spaces.Space:
        """观察空间"""
        ...

    @property
    @abstractmethod
    def action_space(self) -> spaces.Space:
        """动作空间"""
        ...

    @abstractmethod
    def reset(
        self,
        seed: Optional[int] = None,
        options: Optional[Dict] = None
    ) -> Tuple[np.ndarray, Dict]:
        """重置环境

        Returns:
            observation: 初始观察
            info: 额外信息
        """
        ...

    @abstractmethod
    def step(
        self,
        action: np.ndarray
    ) -> Tuple[np.ndarray, float, bool, bool, Dict]:
        """执行一步

        Args:
            action: 动作

        Returns:
            observation: 新观察
            reward: 奖励
            terminated: 是否完成任务
            truncated: 是否提前终止
            info: 额外信息
        """
        ...

    def render(self, mode: str = "human") -> Optional[np.ndarray]:
        """渲染环境

        Args:
            mode: "human" 显示窗口, "rgb_array" 返回图像

        Returns:
            图像数组（如果是 rgb_array 模式）
        """
        ...

    def close(self) -> None:
        """关闭环境"""
        ...

4.1.2 EnvWrapper (base/wrapper.py)

class EnvWrapper(BaseEnv):
    """环境包装器基类

    用于修改环境行为而不改变环境本身
    """

    def __init__(self, env: BaseEnv):
        self.env = env

    @property
    def observation_space(self):
        return self.env.observation_space

    @property
    def action_space(self):
        return self.env.action_space

    def reset(self, seed=None, options=None):
        return self.env.reset(seed, options)

    def step(self, action):
        return self.env.step(action)

    def render(self, mode="human"):
        return self.env.render(mode)

    def close(self):
        return self.env.close()

4.1.3 VectorEnv (base/vector.py)

class VectorEnv(ABC):
    """向量化环境基类

    支持并行运行多个环境实例，提高训练效率
    """

    @property
    @abstractmethod
    def num_envs(self) -> int:
        """环境数量"""
        ...

    @abstractmethod
    def reset(self, seed: Optional[int] = None) -> np.ndarray:
        """重置所有环境"""
        ...

    @abstractmethod
    def step(self, actions: np.ndarray) -> Tuple:
        """在所有环境中执行动作

        Returns:
            observations: (num_envs, *obs_shape)
            rewards: (num_envs,)
            terminateds: (num_envs,)
            truncateds: (num_envs,)
            infos: List[Dict]
        """
        ...

    @abstractmethod
    def close(self) -> None:
        """关闭所有环境"""
        ...

4.2 机器人环境基类 (base/robot_env.py)

RobotEnv 类

class RobotEnv(BaseEnv):
    """机器人仿真环境基类

    扩展 BaseEnv，添加机器人特有的功能：
    - 机器人模型加载
    - 相机渲染
    - 物理交互
    - 奖励计算
    """

    def __init__(
        self,
        task: str,
        robot_type: str = "so100",
        render_mode: str | None = None,
        simulation_backend: str = "mujoco",
        headless: bool = False,
        **kwargs
    ):
        """
        Args:
            task: 任务类型 ("push", "pick_place", "reach")
            robot_type: 机器人类型 ("so100", "aloha", "koch")
            render_mode: 渲染模式 ("human", "rgb_array", None)
            simulation_backend: 仿真后端 ("mujoco", "isaacgym", "pybullet", "sapien")
            headless: 无头模式（不显示窗口）
        """
        ...

    @abstractmethod
    def _load_robot_model(self) -> None:
        """加载机器人模型"""
        ...

    @abstractmethod
    def _setup_cameras(self) -> None:
        """设置相机"""
        ...

    @abstractmethod
    def _compute_reward(self, achieved_goal, desired_goal) -> float:
        """计算奖励"""
        ...

    def _check_termination(self) -> bool:
        """检查是否终止"""
        ...

    @property
    @abstractmethod
    def robot_state(self) -> np.ndarray:
        """获取机器人状态"""
        ...

    @property
    def camera_images(self) -> dict[str, np.ndarray]:
        """获取所有相机图像"""
        ...

class PushEnv(RobotEnv):
    """推任务环境

    目标：用机器人末端执行器推动物体到目标位置
    """

    def __init__(
        self,
        robot_type: str = "so100",
        object_type: str = "cube",
        render_mode: str | None = None,
        simulation_backend: str = "mujoco",
        **kwargs
    ):
        """
        Args:
            robot_type: 机器人类型
            object_type: 被推物体类型 ("cube", "sphere", "cylinder")
        """
        self.object_type = object_type
        super().__init__(
            task="push",
            robot_type=robot_type,
            render_mode=render_mode,
            simulation_backend=simulation_backend,
            **kwargs
        )

    def reset(self, seed: int | None = None):
        """重置环境

        随机放置：
        - 机器人初始位置
        - 物体初始位置
        - 目标位置
        """
        ...

    def step(self, action: np.ndarray):
        """执行一步

        Args:
            action: 机器人动作 (位置控制或力矩控制)

        Returns:
            observation: 包含机器人状态、物体位置、目标位置
            reward: 距离目标的负值（越近奖励越高）
            terminated: 是否成功到达目标
            truncated: 是否超时
            info: 额外信息
        """
        ...

    def _compute_reward(self, achieved_goal, desired_goal):
        """奖励 = -距离目标的距离"""
        return -np.linalg.norm(achieved_goal - desired_goal)

    def _check_termination(self):
        """成功条件：物体与目标距离 < 阈值"""
        return np.linalg.norm(self.object_pos - self.goal_pos) < 0.05

4.3.2 抓取放置任务 (tasks/pick_place.py)

class PickPlaceEnv(RobotEnv):
    """抓取-放置任务环境

    目标：抓起物体并放置到目标位置
    """

    def __init__(
        self,
        robot_type: str = "so100",
        object_type: str = "cube",
        gripper_type: str = "parallel",
        render_mode: str | None = None,
        simulation_backend: str = "mujoco",
        **kwargs
    ):
        """
        Args:
            robot_type: 机器人类型
            object_type: 被抓取物体类型
            gripper_type: 夹爪类型 ("parallel", "suction")
        """
        self.object_type = object_type
        self.gripper_type = gripper_type
        super().__init__(
            task="pick_place",
            robot_type=robot_type,
            render_mode=render_mode,
            simulation_backend=simulation_backend,
            **kwargs
        )

    def reset(self, seed: int | None = None):
        """重置环境

        随机放置：
        - 机器人初始位置
        - 物体初始位置（在桌子表面）
        - 目标放置位置（在桌子表面）
        """
        ...

    def step(self, action: np.ndarray):
        """执行一步

        Returns:
            observation: 包含机器人状态、物体位置、目标位置、夹爪状态
            reward: 综合奖励（接近物体 + 抓取成功 + 接近目标）
            terminated: 是否成功完成
            truncated: 是否超时
            info: 包含 "is_grasped", "is_placed" 等信息
        """
        ...

    def _compute_reward(self, achieved_goal, desired_goal):
        """综合奖励函数"""
        # 接近物体奖励
        gripper_to_obj = np.linalg.norm(self.gripper_pos - self.object_pos)
        reach_reward = -gripper_to_obj

        # 抓取奖励（物体被抬起）
        grasp_reward = 1.0 if self.object_pos[2] > 0.1 else 0.0

        # 放置奖励（接近目标）
        place_reward = -np.linalg.norm(self.object_pos - self.goal_pos)

        # 加权组合
        total_reward = (
            0.3 * reach_reward +
            0.3 * grasp_reward +
            0.4 * place_reward
        )

        return total_reward

    def _check_termination(self):
        """成功条件：物体被放置在目标附近"""
        object_on_table = self.object_pos[2] < 0.05
        near_goal = np.linalg.norm(self.object_pos[:2] - self.goal_pos[:2]) < 0.05
        return object_on_table and near_goal

4.5.5 ManiSkill 后端 (backends/maniskill.py)

class ManiSkillBackend(SimulationBackend):
    """ManiSkill 仿真后端

    特点：
    - 基于 SAPIEN 引擎
    - 专业的机器人操作仿真
    - 支持复杂物理交互
    - 丰富的操作任务库
    - GPU 加速渲染和物理
    - 支持并行仿真
    """

    def __init__(
        self,
        task: str = "PickCube",
        num_envs: int = 1,
        enable_visual_obs: bool = True,
        **kwargs
    ):
        """
        Args:
            task: ManiSkill 任务名称
                - PickCube: 抓取立方体
                - PushCube: 推动立方体
                - StackCube: 堆叠立方体
                - PlugCharger: 插入充电器
                - TurnFaucet: 转动水龙头
                - OpenCabinetDrawer: 打开抽屉
            num_envs: 并行环境数量
            enable_visual_obs: 启用视觉观察
        """
        try:
            import maniskill2_api
            import sapien.core as sapien
        except ImportError:
            raise ImportError(
                "ManiSkill requires maniskill2-api and sapien. "
                "Install with: pip install maniskill2-api sapien"
            )

        self.task = task
        self.num_envs = num_envs

        # 创建 ManiSkill 环境
        self.env = maniskill2_api.create_environment(
            task=task,
            num_envs=num_envs,
            enable_visual_obs=enable_visual_obs,
            **kwargs
        )

        # 获取环境信息
        self.observation_space = self.env.observation_space
        self.action_space = self.env.action_space

    def reset(self, seed: int | None = None) -> tuple:
        """重置环境

        Returns:
            observation: 观察数据
                - 包含机器人关节位置、速度
                - 物体位置、姿态
                - 可选：RGB-D 图像
            info: 额外信息
        """
        obs = self.env.reset()
        return obs, {}

    def step(self, action: np.ndarray) -> tuple:
        """执行一步

        Args:
            action: 动作（机器人关节位置或速度）

        Returns:
            observation: 新观察
            reward: 奖励
            terminated: 是否完成任务
            truncated: 是否超时
            info: 额外信息
        """
        obs, reward, terminated, info = self.env.step(action)
        truncated = False  # ManiSkill 使用 terminated
        return obs, reward, terminated, truncated, info

    def render(self, mode: str = "rgb_array") -> np.ndarray | None:
        """渲染环境

        Args:
            mode: "rgb_array" 返回图像数组，"human" 显示窗口

        Returns:
            图像数组（如果是 rgb_array 模式）
        """
        if mode == "rgb_array":
            return self.env.render(mode="cameras")
        elif mode == "human":
            self.env.render(mode="viewer")
            return None

    def close(self) -> None:
        """关闭环境"""
        self.env.close()

    def get_state(self) -> dict:
        """获取环境状态"""
        return {
            "observation_space": self.observation_space,
            "action_space": self.action_space,
            "num_envs": self.num_envs,
        }

    @property
    def supported_tasks(self) -> list[str]:
        """获取支持的任务列表"""
        return [
            "PickCube",
            "PushCube",
            "StackCube",
            "PlugCharger",
            "TurnFaucet",
            "OpenCabinetDrawer",
        ]

4.6.1 录制包装器 (wrappers/recorder.py)

class RecordVideo(EnvWrapper):
    """录制环境视频"""

    def __init__(
        self,
        env: BaseEnv,
        video_folder: str,
        video_length: int = 500,
        name_prefix: str = "rl-video"
    ):
        """
        Args:
            env: 被包装的环境
            video_folder: 视频保存路径
            video_length: 每个视频的帧数
            name_prefix: 视频文件名前缀
        """
        super().__init__(env)
        self.video_folder = Path(video_folder)
        self.video_length = video_length
        self.name_prefix = name_prefix
        self.frames = []
        self.step_count = 0

    def step(self, action):
        obs, reward, terminated, truncated, info = self.env.step(action)

        # 记录帧
        frame = self.env.render("rgb_array")
        self.frames.append(frame)

        # 保存视频
        if len(self.frames) >= self.video_length:
            self._save_video()

        return obs, reward, terminated, truncated, info

    def _save_video(self):
        """保存视频到磁盘"""
        from waleo_utils.io import write_video
        video_path = self.video_folder / f"{self.name_prefix}_{self.step_count}.mp4"
        write_video(video_path, np.array(self.frames), fps=30)
        self.frames = []
        self.step_count += 1

4.6.2 图像处理包装器 (wrappers/grayscale.py, wrappers/resize.py)

class GrayScaleObservation(EnvWrapper):
    """将彩色图像转换为灰度图"""

    def __init__(self, env: BaseEnv, keep_dim: bool = False):
        super().__init__(env)
        self.keep_dim = keep_dim

    def observation(self, obs):
        # 转换图像
        if "observation.image" in obs:
            img = obs["observation.image"]
            gray = np.mean(img, axis=-1, keepdims=self.keep_dim)
            obs["observation.image"] = gray
        return obs

class ResizeObservation(EnvWrapper):
    """调整观察图像大小"""

    def __init__(self, env: BaseEnv, shape: tuple[int, int]):
        super().__init__(env)
        self.shape = shape

    def observation(self, obs):
        if "observation.image" in obs:
            import cv2
            img = obs["observation.image"]
            resized = cv2.resize(img, self.shape)
            obs["observation.image"] = resized
        return obs

4.8.2 奖励计算 (utils/reward.py)

def compute_distance_reward(
    achieved: np.ndarray,
    desired: np.ndarray,
    threshold: float = 0.05
) -> float:
    """计算距离奖励

    Returns:
        奖励值（负距离，在阈值内返回0）
    """
    dist = np.linalg.norm(achieved - desired)
    return max(-dist, 0.0) if dist < threshold else -dist

def compute_sparse_reward(
    achieved: np.ndarray,
    desired: np.ndarray,
    threshold: float = 0.05
) -> float:
    """计算稀疏奖励（仅成功时给奖励）

    Returns:
        1.0 if success, else 0.0
    """
    dist = np.linalg.norm(achieved - desired)
    return 1.0 if dist < threshold else 0.0
